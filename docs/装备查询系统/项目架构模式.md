### 1. 实体模式 (Entity Pattern)

#### 基础实体类
```csharp
using System;
using System.ComponentModel.DataAnnotations;
using System.ComponentModel.DataAnnotations.Schema;
using Abp.Domain.Entities.Auditing;
using Abp.AutoMapper;

namespace TtWork.Project.Domains;

[Table("T_Equipment")]
public class Equipment : FullAuditedAggregateRoot<long>
{
    [StringLength(128)]
    public string Name { get; set; }

    [StringLength(64)]
    public string Type { get; set; }  // 武器、防具、首饰、属性水晶

    [StringLength(64)]
    public string SubType { get; set; }  // 剑类、杖类等

    public int Level { get; set; }

    [StringLength(64)]
    public string Quality { get; set; }  // 普通、优秀、精良、传说

    // 属性字段
    public int? Attack { get; set; }
    public int? Defense { get; set; }
    public int? MagicAttack { get; set; }
    public int? MagicDefense { get; set; }
    public int? Agility { get; set; }
    public int? HitRate { get; set; }
    public int? DodgeRate { get; set; }

    public string SpecialEffect { get; set; }

    [StringLength(256)]
    public string ImageUrl { get; set; }

    public string Description { get; set; }

    // JSON字段存储复杂对象
    public string Requirements { get; set; }  // 适用职业、等级要求等
    public string Synthesis { get; set; }     // 合成配方
    public string DropLocations { get; set; } // 掉落位置

    [StringLength(128)]
    public string Source { get; set; }  // 数据来源
}
```

### 3. ApplicationService 模式

```csharp
using System;
using System.Linq;
using System.Threading.Tasks;
using Abp.Application.Services.Dto;
using Abp.Authorization;
using Abp.Domain.Repositories;
using Abp.Linq.Extensions;
using TtWork.Project.Domains;
using TtWork.Project.Applications.Equipments.Dto;

namespace TtWork.Project.Applications.Equipments;

[AbpAuthorize]
public class EquipmentAppService : AsyncCrudAppService<
    Equipment,
    EquipmentDto,
    long,
    PagedEquipmentResultRequestDto,
    CreateOrUpdateEquipmentDto,
    CreateOrUpdateEquipmentDto>
{
    private readonly IRepository<Equipment, long> _equipmentRepository;

    public EquipmentAppService(IRepository<Equipment, long> repository)
        : base(repository)
    {
        _equipmentRepository = repository;
    }

    protected override IQueryable<Equipment> CreateFilteredQuery(PagedEquipmentResultRequestDto input)
    {
        var query = base.CreateFilteredQuery(input);

        // 关键词搜索
        if (!input.Keyword.IsNullOrWhiteSpace())
        {
            query = query.Where(e =>
                e.Name.Contains(input.Keyword) ||
                e.SubType.Contains(input.Keyword) ||
                e.Description.Contains(input.Keyword));
        }

        // 类型筛选
        if (!input.Type.IsNullOrWhiteSpace())
        {
            query = query.Where(e => e.Type == input.Type);
        }

        // 子类型筛选
        if (!input.SubType.IsNullOrWhiteSpace())
        {
            query = query.Where(e => e.SubType == input.SubType);
        }

        // 品质筛选
        if (!input.Quality.IsNullOrWhiteSpace())
        {
            query = query.Where(e => e.Quality == input.Quality);
        }

        // 等级范围
        if (input.MinLevel.HasValue)
        {
            query = query.Where(e => e.Level >= input.MinLevel);
        }

        if (input.MaxLevel.HasValue)
        {
            query = query.Where(e => e.Level <= input.MaxLevel);
        }

        return query;
    }

    protected override IQueryable<Equipment> ApplySorting(IQueryable<Equipment> query, PagedEquipmentResultRequestDto input)
    {
        if (input.Sorting.IsNullOrWhiteSpace())
        {
            return base.ApplySorting(query, input);
        }

        return query.OrderBy(input.Sorting);
    }

    public override async Task<PagedResultDto<EquipmentDto>> GetAll(PagedEquipmentResultRequestDto input)
    {
        return await base.GetAll(input);
    }

    public override async Task<EquipmentDto> Get(EntityDto<long> input)
    {
        return await base.Get(input);
    }

    public override async Task<EquipmentDto> Create(CreateOrUpdateEquipmentDto input)
    {
        return await base.Create(input);
    }

    public override async Task<EquipmentDto> Update(CreateOrUpdateEquipmentDto input)
    {
        return await base.Update(input);
    }

    public override async Task Delete(EntityDto<long> input)
    {
        await base.Delete(input);
    }

    /// <summary>
    /// 批量创建装备
    /// </summary>
    public async Task BatchCreate(CreateOrUpdateEquipmentDto[] inputs)
    {
        foreach (var input in inputs)
        {
            await Create(input);
        }
    }

    /// <summary>
    /// 获取装备类型统计
    /// </summary>
    public async Task<object> GetTypeStatistics()
    {
        var query = await _equipmentRepository.GetAllListAsync();

        var result = query
            .GroupBy(e => e.Type)
            .Select(g => new
            {
                Type = g.Key,
                Count = g.Count()
            })
            .OrderByDescending(g => g.Count)
            .ToList();

        return result;
    }
}
```

### 4. 验证模式 (Validation Pattern)

```csharp
using FluentValidation;
using TtWork.Lib;

namespace TtWork.Project.Applications.Equipments.Dto;

public class CreateOrUpdateEquipmentDtoValidator : AbstractValidator<CreateOrUpdateEquipmentDto>
{
    public CreateOrUpdateEquipmentDtoValidator()
    {
        RuleFor(x => x).Must(x => !x.Name.IsNullOrEmptyOrWhiteSpace())
            .WithMessage("装备名称不能为空");

        RuleFor(x => x).Must(x => !x.Type.IsNullOrEmptyOrWhiteSpace())
            .WithMessage("装备类型不能为空");

        RuleFor(x => x).Must(x => !x.SubType.IsNullOrEmptyOrWhiteSpace())
            .WithMessage("装备子类型不能为空");

        RuleFor(x => x.Level)
            .GreaterThan(0)
            .LessThanOrEqualTo(9)
            .WithMessage("装备等级必须在1-9之间");

        RuleFor(x => x).Must(x => !x.ImageUrl.IsNullOrEmptyOrWhiteSpace())
            .WithMessage("装备图片不能为空");

        RuleFor(x => x.Attack)
            .GreaterThanOrEqualTo(0)
            .When(x => x.Attack.HasValue)
            .WithMessage("攻击力不能为负数");

        RuleFor(x => x.Defense)
            .GreaterThanOrEqualTo(0)
            .When(x => x.Defense.HasValue)
            .WithMessage("防御力不能为负数");
    }
}
```

### 3. Pinia Store 模式

```typescript
// stores/equipment.ts
import { defineStore } from 'pinia'
import { ref, computed } from 'vue'
import { equipmentApi } from '@/api/equipment'
import type {
  Equipment,
  EquipmentSearchParams,
  PagedEquipmentResult
} from '@/types/equipment'

export const useEquipmentStore = defineStore('equipment', () => {
  // State
  const equipmentList = ref<Equipment[]>([])
  const selectedEquipment = ref<Equipment | null>(null)
  const compareList = ref<Equipment[]>([])
  const totalCount = ref(0)
  const loading = ref(false)

  // Getters
  const canCompareMore = computed(() => compareList.value.length < 3)
  const selectedTypes = computed(() => {
    return Array.from(new Set(equipmentList.value.map(e => e.type)))
  })

  // Actions
  const searchEquipments = async (params: EquipmentSearchParams) => {
    loading.value = true
    try {
      const result: PagedEquipmentResult = await equipmentApi.getList(params)
      equipmentList.value = result.items
      totalCount.value = result.totalCount
      return result
    } finally {
      loading.value = false
    }
  }

  const getEquipmentDetail = async (id: number) => {
    loading.value = true
    try {
      const result = await equipmentApi.getDetail(id)
      selectedEquipment.value = result
      return result
    } finally {
      loading.value = false
    }
  }

  const addToCompare = (equipment: Equipment) => {
    if (!canCompareMore.value) {
      throw new Error('最多只能对比3件装备')
    }
    if (compareList.value.find(e => e.id === equipment.id)) {
      throw new Error('该装备已在对比列表中')
    }
    compareList.value.push(equipment)
  }

  const removeFromCompare = (id: number) => {
    compareList.value = compareList.value.filter(e => e.id !== id)
  }

  const clearCompare = () => {
    compareList.value = []
  }

  const resetSearch = () => {
    equipmentList.value = []
    totalCount.value = 0
    selectedEquipment.value = null
  }

  return {
    // State
    equipmentList,
    selectedEquipment,
    compareList,
    totalCount,
    loading,
    // Getters
    canCompareMore,
    selectedTypes,
    // Actions
    searchEquipments,
    getEquipmentDetail,
    addToCompare,
    removeFromCompare,
    clearCompare,
    resetSearch
  }
})
```

## 🗄️ 数据库设计模式

### 表命名规范
- 表名前缀: `T_`
- PascalCase命名: `T_Equipment`, `T_AuctionItem`

### 字段命名规范
- PascalCase: `Name`, `ImageUrl`, `Attack`
- 外键: `XXXId` (如 `SellerId`)
- 布尔值: `IsXXX` (如 `IsActive`)

### 数据库迁移命名
- 格式: `YYYYMMDD_Description`
- 示例: `20260307_CreateEquipmentTable`

### 索引设计
```sql
-- 主键索引
PRIMARY KEY (Id)

-- 常用查询字段索引
CREATE INDEX IX_Equipment_Type ON T_Equipment(Type);
CREATE INDEX IX_Equipment_Level ON T_Equipment(Level);
CREATE INDEX IX_Equipment_Quality ON T_Equipment(Quality);

-- 复合索引
CREATE INDEX IX_Equipment_Type_Level ON T_Equipment(Type, Level);
CREATE INDEX IX_Equipment_Name ON T_Equipment(Name);

-- 全文搜索索引
CREATE FULLTEXT INDEX IX_Equipment_FullText ON T_Equipment(Name, Description);
```
