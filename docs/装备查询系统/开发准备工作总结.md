#### ApplicationService 模式
```csharp
[AbpAuthorize]
public class EquipmentAppService : AsyncCrudAppService<Equipment, EquipmentDto, long, PagedEquipmentResultRequestDto, CreateEquipmentDto, UpdateEquipmentDto>
{
    private readonly IRepository<Equipment, long> _equipmentRepository;

    public EquipmentAppService(IRepository<Equipment, long> repository)
        : base(repository)
    {
        _equipmentRepository = repository;
    }

    protected override IQueryable<Equipment> CreateFilteredQuery(PagedEquipmentResultRequestDto input)
    {
        var query = base.CreateFilteredQuery(input);

        if (!input.Type.IsNullOrWhiteSpace())
        {
            query = query.Where(e => e.Type == input.Type);
        }

        if (!input.SubType.IsNullOrWhiteSpace())
        {
            query = query.Where(e => e.SubType == input.SubType);
        }

        if (input.MinLevel.HasValue)
        {
            query = query.Where(e => e.Level >= input.MinLevel);
        }

        if (input.MaxLevel.HasValue)
        {
            query = query.Where(e => e.Level <= input.MaxLevel);
        }

        return query;
    }
}
```

#### Pinia Store 模式
```typescript
import { defineStore } from 'pinia'
import { equipmentApi } from '@/api/equipment'

export const useEquipmentStore = defineStore('equipment', () => {
  const equipmentList = ref([])
  const selectedEquipment = ref(null)
  const compareList = ref([])

  const searchEquipments = async (params: EquipmentSearchParams) => {
    const result = await equipmentApi.search(params)
    equipmentList.value = result.items
    return result
  }

  const getEquipmentDetail = async (id: number) => {
    const result = await equipmentApi.getDetail(id)
    selectedEquipment.value = result
    return result
  }

  const addToCompare = (equipment: Equipment) => {
    if (compareList.value.length >= 3) {
      throw new Error('最多只能对比3件装备')
    }
    if (compareList.value.find(e => e.id === equipment.id)) {
      throw new Error('该装备已在对比列表中')
    }
    compareList.value.push(equipment)
  }

  const removeFromCompare = (id: number) => {
    compareList.value = compareList.value.filter(e => e.id !== id)
  }

  const clearCompare = () => {
    compareList.value = []
  }

  return {
    equipmentList,
    selectedEquipment,
    compareList,
    searchEquipments,
    getEquipmentDetail,
    addToCompare,
    removeFromCompare,
    clearCompare
  }
})
```
